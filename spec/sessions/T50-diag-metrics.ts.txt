import { writeFileSync } from 'node:fs';
import { compile } from '/home/aleksa/pmov_generator/src/lang/index.ts';
import { buildDiagramGraph, estimateTextWidth } from '/home/aleksa/pmov_generator/src/render/graph/index.ts';
import { cardBoxes, layoutDiagram } from '/home/aleksa/pmov_generator/src/render/layout/index.ts';
import { boxOf, computeDiagramGeometry } from '/home/aleksa/pmov_generator/src/render/geometry/index.ts';
import { graphToSvg } from '/home/aleksa/pmov_generator/src/render/svg/index.ts';
import { EXAMPLES } from '/home/aleksa/pmov_generator/src/examples/index.ts';
type R = { x: number; y: number; w: number; h: number };
const ov = (a: R, b: R) => Math.max(0, Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x)) * Math.max(0, Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y));
for (const ex of EXAMPLES) {
  const g = buildDiagramGraph(compile(ex.source).ir!);
  const pos = await layoutDiagram(g, { algorithm: 'grid' });
  if (ex.id === 'proizvodnja') writeFileSync('/home/aleksa/pmov_generator/spec/sessions/T46-proizvodnja-grid.svg', graphToSvg(g, pos));
  const boxes = Object.fromEntries(g.nodes.map((n) => [n.id, boxOf(n, pos[n.id]!)]));
  const geo = computeDiagramGeometry(g, boxes);
  const cards = Object.entries(cardBoxes(g, pos));
  let cardOv = 0; for (let i = 0; i < cards.length; i++) for (let j = i + 1; j < cards.length; j++) { const a = cards[i]![1], b = cards[j]![1]; if (ov({ x: a.x, y: a.y, w: a.width, h: a.height }, { x: b.x, y: b.y, w: b.width, h: b.height }) > 1) cardOv++; }
  let slanted = 0;
  const labels: { e: string; r: R }[] = [];
  for (const e of g.edges) {
    const x = geo[e.id]!; const t = g.nodes.find((n) => n.id === e.target)!;
    if (t.type === 'atribut') { const a = x.points[0]!, b = x.points[x.points.length - 1]!; if (Math.abs(a.y - b.y) > 0.5) slanted++; }
    if (x.label && e.label) { const w = estimateTextWidth(e.label, 12); const lx = x.label.anchor === 'start' ? x.label.x : x.label.anchor === 'end' ? x.label.x - w : x.label.x - w / 2; labels.push({ e: `${e.id} "${e.label}"`, r: { x: lx, y: x.label.y - 7, w, h: 14 } }); }
  }
  const hits: string[] = [];
  for (let i = 0; i < labels.length; i++) for (let j = i + 1; j < labels.length; j++) { const a = ov(labels[i]!.r, labels[j]!.r); if (a > 1) hits.push(`label ${labels[i]!.e} × label ${labels[j]!.e} (${a.toFixed(0)}px²)`); }
  for (const l of labels) for (const n of g.nodes) { const b = boxes[n.id]!; const a = ov(l.r, { x: b.x, y: b.y, w: b.width, h: b.height }); if (a > 1) hits.push(`label ${l.e} × ${n.type} ${n.id} (${a.toFixed(0)}px²)`); }
  console.log(`${ex.id}: cardOverlaps=${cardOv} slantedStubs=${slanted}`); for (const h of hits) console.log('   ' + h);
}
